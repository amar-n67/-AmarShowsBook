AmarShowsBook SQL Query Reference
Database: AmarMovie_DB
Dialect: PostgreSQL

Use these as copy-ready templates for inserting, updating, deleting, and viewing the main application data.

============================================================
MOVIES
============================================================

-- View movies
SELECT * FROM public."Movies" ORDER BY "Id" DESC;

-- Insert movie
INSERT INTO public."Movies"
("Title", "Director", "Producer", "Cast", "Duration", "PosterUrl", "Images", "TrailerUrl", "Description", "ImdbRating")
VALUES
('Sample Movie', 'Director Name', 'Producer Name', 'Actor One, Actor Two', 150, '/uploads/movie-poster.jpg', '/uploads/movie-1.jpg,/uploads/movie-2.jpg', 'https://www.youtube.com/watch?v=VIDEO_ID', 'Movie description', 8.2);

-- Update movie
UPDATE public."Movies"
SET "Title" = 'Updated Movie',
    "Director" = 'Updated Director',
    "Duration" = 145,
    "ImdbRating" = 8.5
WHERE "Id" = 1;

-- Delete movie
DELETE FROM public."Movies" WHERE "Id" = 1;
