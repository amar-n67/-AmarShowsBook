showTime SQL query notes
Updated: 2026-08-24
Database: AmarMovie_DB
Database type: PostgreSQL

Purpose
This file keeps useful SQL in one place.
Use it when you need to check data, fix a record, test a page, or understand which table/view feeds a screen.

Important safety rule
- Read data first with SELECT.
- Update or delete only with a clear WHERE condition.
- Replace sample ids like 1, 23, or 71 with the real id you want.
- Do not create custom roles. The app supports only AMAR_SUPER_ADMIN, AMAR_ADMIN, AMAR_DEVELOPER, and AMAR_USER.

============================================================
USERS, ROLES, ACCESS, AND MENUS
============================================================

Users

SELECT "Id", "Name", "Email", is_active, is_deleted
FROM public."Users"
ORDER BY "Id" DESC;

Standard roles

SELECT id, role_code, role_name, is_active
FROM public.roles
ORDER BY id;

User role mappings

SELECT urm.user_id, u."Email", r.role_code, r.role_name, urm.is_active
FROM public.user_role_mappings urm
JOIN public."Users" u ON u."Id" = urm.user_id
JOIN public.roles r ON r.id = urm.role_id
ORDER BY u."Id", r.role_code;

Access matrix

SELECT user_id, user_email, role_code, module_code, action_type, permission_code
FROM public.vw_user_access_matrix
ORDER BY user_id, module_code, action_type;

Menus visible to users

SELECT user_id, role_code, menu_code, menu_name, route_path, can_view, can_create, can_update, can_delete
FROM public.vw_user_application_menus
ORDER BY user_id, display_order;

Give one of the four roles to a user

INSERT INTO public.user_role_mappings
(user_id, role_id, assigned_by, assigned_at, is_active)
SELECT u."Id", r.id, NULL, CURRENT_TIMESTAMP, true
FROM public."Users" u
JOIN public.roles r ON r.role_code = 'AMAR_ADMIN'
WHERE u."Email" = 'admin@example.com'
  AND NOT EXISTS (
      SELECT 1
      FROM public.user_role_mappings existing
      WHERE existing.user_id = u."Id"
        AND existing.role_id = r.id
        AND existing.is_active = true
  );

Disable a user's role mapping

UPDATE public.user_role_mappings
SET is_active = false
WHERE user_id = 1
  AND role_id = (
      SELECT id
      FROM public.roles
      WHERE role_code = 'AMAR_ADMIN'
  );

Find a user's bookings

SELECT b.*
FROM public.bookings b
JOIN public."Users" u ON u."Id" = b.user_id
WHERE lower(u."Email") = lower('user@example.com')
ORDER BY b.id DESC;

Mark a booking confirmed

UPDATE public.bookings
SET booking_status = 'CONFIRMED',
    payment_status = 'SUCCESS',
    updated_at = CURRENT_TIMESTAMP
WHERE id = 1;

Cancel a booking from SQL only when you know this is the correct record

UPDATE public.bookings
SET booking_status = 'CANCELLED',
    payment_status = 'REFUND_PENDING',
    cancelled_at = CURRENT_TIMESTAMP,
    cancellation_reason = 'Admin cancelled'
WHERE id = 1;
